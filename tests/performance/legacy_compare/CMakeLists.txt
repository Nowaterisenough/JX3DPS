function(configure_legacy TARGET)
    target_include_directories(${TARGET} PRIVATE "${LEGACY_ROOT}/src" "${LEGACY_ROOT}/include"
        "${LEGACY_ROOT}/modules" "${LEGACY_ROOT}/3rd/spdlog/include" "${LEGACY_ROOT}/3rd/json/include")
    target_compile_definitions(${TARGET} PRIVATE EXPORT_JX3DPS EXPORT_THREAD_POOL VERSION_PATCH=0 VERSION_TWEAK=0 VERSION_BRANCH=benchmark)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # Bundled fmt predates Clang's stricter consteval checks in modern modes.
        target_compile_definitions(${TARGET} PRIVATE FMT_CONSTEVAL=)
    endif()
    if(MSVC)
        target_compile_options(${TARGET} PRIVATE /utf-8 /FI${CMAKE_CURRENT_SOURCE_DIR}/compat.hpp)
    else()
        target_compile_options(${TARGET} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/compat.hpp")
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            target_compile_options(${TARGET} PRIVATE -Wno-braced-scalar-init)
        endif()
    endif()
    target_link_libraries(${TARGET} PRIVATE Threads::Threads)
endfunction()

option(BUILD_DPS_DIAGNOSTIC "Also measure a modified legacy engine without attribute gain calculations" ON)
if(BUILD_DPS_DIAGNOSTIC)
    foreach(NAME Skill Buff)
        file(READ "${LEGACY_ROOT}/src/${NAME}.cpp" SOURCE)
        set(TYPE_ARRAY "std::array<Attribute::Type, 6> types = \\{[^;]*;")
        string(REGEX MATCHALL "std::array<Attribute::Type, 6> types" ARRAYS "${SOURCE}")
        list(LENGTH ARRAYS ARRAY_COUNT)
        if(NOT ARRAY_COUNT EQUAL 4)
            message(FATAL_ERROR "Unexpected legacy ${NAME}.cpp; verify the pinned revision")
        endif()
        string(REGEX REPLACE "${TYPE_ARRAY}" "std::array<Attribute::Type, 1> types = {Attribute::Type::DEFAULT};" SOURCE "${SOURCE}")
        string(REGEX REPLACE "gainsDamage\\[Attribute::Type::[A-Z_]+\\] = gainsDamage\\[Attribute::Type::DEFAULT\\];" "" SOURCE "${SOURCE}")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${NAME}.dps.cpp" "${SOURCE}")
    endforeach()
    add_executable(legacy_dps_bench legacy_bench.cpp "${CMAKE_CURRENT_BINARY_DIR}/Skill.dps.cpp" "${CMAKE_CURRENT_BINARY_DIR}/Buff.dps.cpp" $<TARGET_OBJECTS:legacy_common>)
