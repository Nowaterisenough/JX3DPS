if(NOT EXISTS "${PROJECT_SOURCE_DIR}/3rd/json/single_include/nlohmann/json.hpp")
    message(FATAL_ERROR "Initialize the engine dependency with: git submodule update --init 3rd/json")
endif()
if(JX3DPS_SHARED_ENGINE AND NOT EMSCRIPTEN)
    add_library(JX3DPS SHARED engine/engine.cpp)
    target_compile_definitions(JX3DPS PUBLIC JX3DPS_SHARED PRIVATE JX3DPS_BUILD)
    set_target_properties(JX3DPS PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
else()
    add_library(JX3DPS STATIC engine/engine.cpp)
endif()
set(permanent_catalog "${PROJECT_SOURCE_DIR}/apps/desktop/data/permanents50.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${permanent_catalog}")
file(READ "${permanent_catalog}" PERMANENTS_JSON)
# Keep each literal below MSVC's individual string-literal size limit.
string(REPLACE "\n" ")jx3\"\nR\"jx3(" PERMANENTS_JSON "${PERMANENTS_JSON}")
configure_file(class/common/permanents_data.hpp.in generated/permanents_data.hpp @ONLY)
# Each school exports a Book factory with the same Chinese name as its source file.
file(GLOB school_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/class/*.cpp")
set(school_factories "")
foreach(source IN LISTS school_sources)
    get_filename_component(school "${source}" NAME_WE)
    string(APPEND school_factories "JX3DPS_SCHOOL(${school})\n")
    if(MINGW)
        # Some Windows binutils code pages collapse different Chinese object names at link time.
        string(SHA256 school_id "${school}")
        set(unit "${CMAKE_CURRENT_BINARY_DIR}/generated/school_${school_id}.cpp")
        file(GENERATE OUTPUT "${unit}" CONTENT "#include \"src/class/${school}.cpp\"\n")
        list(APPEND school_units "${unit}")
    else()
        list(APPEND school_units "${source}")
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/school_factories.inc" CONTENT "${school_factories}")
add_library(jx3dps_season50 OBJECT class/common/books.cpp class/common/permanents.cpp ${school_units})
target_include_directories(jx3dps_season50 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${PROJECT_SOURCE_DIR}/3rd/json/single_include")
target_link_libraries(jx3dps_season50 PUBLIC jx3dps_runtime_core)
target_compile_features(jx3dps_season50 PUBLIC cxx_std_23)
set_target_properties(jx3dps_season50 PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(jx3dps_season50 PRIVATE /utf-8)
elseif(EMSCRIPTEN)
    target_compile_options(jx3dps_season50 PRIVATE -fexceptions)
endif()
target_link_libraries(JX3DPS PRIVATE jx3dps_season50)
target_link_libraries(JX3DPS PRIVATE jx3dps_runtime_core)
target_include_directories(JX3DPS PUBLIC "$<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:include>" PRIVATE "${PROJECT_SOURCE_DIR}/3rd/json/single_include")
target_compile_features(JX3DPS PRIVATE cxx_std_23)
if(EMSCRIPTEN)
    target_compile_options(JX3DPS PRIVATE -fexceptions)
else()
    add_library(jx3dps_engine_client STATIC engine/client.cpp)
    target_link_libraries(jx3dps_engine_client PUBLIC JX3DPS jx3dps_runtime_core jx3dps_season50)
    target_include_directories(jx3dps_engine_client PRIVATE "${PROJECT_SOURCE_DIR}/3rd/json/single_include")
    target_compile_features(jx3dps_engine_client PRIVATE cxx_std_23)
    if(MSVC)
        target_compile_options(jx3dps_engine_client PRIVATE /utf-8)
    endif()
    find_package(Threads REQUIRED)
    target_link_libraries(JX3DPS PRIVATE Threads::Threads)
    install(TARGETS JX3DPS ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
    install(FILES "${PROJECT_SOURCE_DIR}/include/jx3dps.h" DESTINATION include)
endif()
if(MSVC)
    target_compile_options(JX3DPS PRIVATE /utf-8)
endif()
