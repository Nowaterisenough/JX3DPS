foreach(target IN ITEMS season50_runtime tai_xu_runtime mo_wen_runtime)
    add_executable(${target} season50_runtime.cpp)
    target_link_libraries(${target} PRIVATE JX3DPS)
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/3rd/json/single_include")
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
endforeach()
target_compile_definitions(tai_xu_runtime PRIVATE JX3DPS_CLI_SCHOOL="tx")
target_compile_definitions(mo_wen_runtime PRIVATE JX3DPS_CLI_SCHOOL="mw")
