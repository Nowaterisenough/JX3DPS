find_package(Threads REQUIRED)
foreach(target IN ITEMS tai_xu_runtime mo_wen_runtime)
    set(source ${target}.cpp)
    set(target legacy_${target})
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE jx3dps_engine_client Threads::Threads)
    if(MSVC AND NOT "cxx_std_26" IN_LIST CMAKE_CXX_COMPILE_FEATURES)
        target_compile_options(${target} PRIVATE /std:c++latest)
    else()
        target_compile_features(${target} PRIVATE cxx_std_26)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

foreach(target IN ITEMS season50_runtime tai_xu_runtime mo_wen_runtime)
    add_executable(${target} season50_runtime.cpp)
    target_link_libraries(${target} PRIVATE JX3DPS)
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/3rd/json/single_include")
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
endforeach()
target_compile_definitions(tai_xu_runtime PRIVATE JX3DPS_CLI_SCHOOL="tx")
target_compile_definitions(mo_wen_runtime PRIVATE JX3DPS_CLI_SCHOOL="mw")
